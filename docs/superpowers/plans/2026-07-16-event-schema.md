# Event Schema Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Create the `event` document type in the Sanity schema and add GROQ queries for event pages.

**Architecture:** A new `eventType` schema file in `sanity/schemaTypes/`, registered in the schema index. Reuses `imageField()`, `linkField()`, `programField()`, and `creditFields` utilities. Two GROQ queries added to `client/src/sanity/queries.ts`: one for a single event by slug, one for upcoming events list.

**Tech Stack:** Sanity v5 (schema), GROQ, TypeScript, next-sanity

## Global Constraints

- All field definitions use `defineField`, root type uses `defineType`
- `_id` values auto-generated by Sanity
- Follow existing patterns: `imageField()` for required images, `programField()` for program references, `linkField()` for link objects, `creditFields` spread for credits
- Sanity icons from `@sanity/icons`
- Queries wrapped in `defineQuery` from `next-sanity`

```typescript
// sanity/schemaTypes/eventType.ts
import { defineField, defineType } from 'sanity'
import { CalendarIcon } from '@sanity/icons'

import { imageField } from '@/util/image'
import { linkField } from '@/util/link'
import { programField } from '@/util/program'
import { creditFields } from '@/schemaTypes/shared/creditFields'

export const eventType = defineType({
  name: 'event',
  title: 'Event',
  type: 'document',
  icon: CalendarIcon,
  fields: [
    // --- Core Metadata ---
    defineField({
      title: 'Title',
      name: 'title',
      type: 'string',
      validation: (rule) => rule.required(),
    }),
    defineField({
      title: 'Slug',
      name: 'slug',
      type: 'slug',
      options: { source: 'title' },
      validation: (rule) => rule.required(),
    }),
    defineField(imageField({
      title: 'Hero Image',
      name: 'heroImage',
    })),
    defineField(programField({
      title: 'Program',
      name: 'program',
    })),
    defineField({
      title: 'Description',
      name: 'description',
      type: 'array',
      of: [{ type: 'block' }],
    }),

    // --- Event Logistics ---
    defineField({
      title: 'Date / Times',
      name: 'dateTimes',
      type: 'array',
      of: [
        defineField({
          type: 'object',
          name: 'dateTimeRange',
          title: 'Date / Time Range',
          fields: [
            defineField({
              title: 'Start',
              name: 'start',
              type: 'datetime',
              validation: (rule) => rule.required(),
            }),
            defineField({
              title: 'End',
              name: 'end',
              type: 'datetime',
              validation: (rule) => rule.required(),
            }),
          ],
          preview: {
            select: { start: 'start', end: 'end' },
            prepare: ({ start, end }) => ({
              title: `${start} \u2013 ${end}`,
            }),
          },
        }),
      ],
    }),
    defineField({
      title: 'Cost',
      name: 'cost',
      type: 'string',
    }),
    defineField({
      title: 'Location',
      name: 'location',
      type: 'text',
    }),
    defineField({
      title: 'Access Info',
      name: 'accessInfo',
      type: 'text',
    }),
    defineField({
      title: 'Links',
      name: 'links',
      type: 'array',
      of: [linkField()],
    }),

    // --- Optional Content Sections ---
    defineField({
      title: 'About the Artists',
      name: 'featuredArtists',
      type: 'array',
      of: [
        defineField({
          type: 'object',
          name: 'featuredArtist',
          title: 'Featured Artist',
          fields: [
            defineField({
              title: 'Artist',
              name: 'artist',
              type: 'reference',
              to: [{ type: 'artist' }],
            }),
            defineField({
              title: 'Image',
              name: 'image',
              type: 'image',
              options: { hotspot: true },
              fields: [
                defineField({
                  name: 'alt',
                  type: 'string',
                  title: 'Alt text',
                }),
              ],
            }),
            defineField({
              title: 'Name',
              name: 'name',
              type: 'string',
            }),
            defineField({
              title: 'Bio',
              name: 'bio',
              type: 'array',
              of: [{ type: 'block' }],
            }),
          ],
          preview: {
            select: { name: 'name', artistName: 'artist.name', media: 'image' },
            prepare: ({ name, artistName, media }) => ({
              title: name || artistName || 'Featured Artist',
              media,
            }),
          },
        }),
      ],
    }),
    defineField({
      title: 'Artworks on View',
      name: 'artworks',
      type: 'array',
      of: [
        defineField({
          type: 'object',
          name: 'artwork',
          title: 'Artwork',
          fields: [
            defineField(imageField({
              title: 'Image',
              name: 'image',
            })),
            defineField({
              title: 'Description',
              name: 'description',
              type: 'array',
              of: [{ type: 'block' }],
            }),
          ],
          preview: {
            select: { media: 'image' },
            prepare: ({ media }) => ({
              title: 'Artwork',
              media,
            }),
          },
        }),
      ],
    }),
    defineField({
      title: 'About the Program',
      name: 'aboutProgram',
      type: 'array',
      of: [{ type: 'block' }],
    }),
    defineField({
      title: 'Schedule',
      name: 'schedule',
      type: 'object',
      fields: [
        defineField({
          title: 'Description',
          name: 'description',
          type: 'array',
          of: [{ type: 'block' }],
        }),
        defineField({
          title: 'Items',
          name: 'items',
          type: 'array',
          of: [
            defineField({
              type: 'object',
              name: 'scheduleItem',
              title: 'Schedule Item',
              fields: [
                defineField({
                  title: 'Title',
                  name: 'title',
                  type: 'string',
                  validation: (rule) => rule.required(),
                }),
                defineField({
                  title: 'Time',
                  name: 'time',
                  type: 'string',
                }),
                defineField({
                  title: 'Description',
                  name: 'description',
                  type: 'array',
                  of: [{ type: 'block' }],
                }),
              ],
              preview: {
                select: { title: 'title', time: 'time' },
                prepare: ({ title, time }) => ({
                  title: title,
                  subtitle: time,
                }),
              },
            }),
          ],
        }),
      ],
    }),

    // --- Credits ---
    ...creditFields,
  ],
})
```

- [ ] **Step 1: Add `getEventBySlug` query**

Append to `client/src/sanity/queries.ts`:

```typescript
export function getEventBySlug(slug: string) {
  const getEventBySlugQuery = defineQuery(
    `*[_type == "event" && slug.current == $slug][0]{
      ...,
      "program": program->{
        _id, title, slug, shortLabel, displayTitle, accentColor
      },
      featuredArtists[]{
        _key,
        artist->{ _id, name, slug },
        "image": coalesce(image, artist->image),
        "name": coalesce(name, artist->name),
        "bio": coalesce(bio, artist->bio)
      }
    }`,
  );

  return client.fetch(getEventBySlugQuery, { slug }, options);
}
```

- [ ] **Step 2: Add `getUpcomingEvents` query**

Append to `client/src/sanity/queries.ts`:

```typescript
export function getUpcomingEvents(limit: number = 10) {
  const now = new Date().toISOString();

  const getUpcomingEventsQuery = defineQuery(
    `*[
      _type == "event"
      && defined(slug.current)
      && count(dateTimes) > 0
      && dateTimes[-1].end >= $now
    ] | order(dateTimes[0].start asc) [0...$limit] {
      _id,
      title,
      "slug": slug.current,
      dateTimes,
      location,
      "program": program->{
        _id, title, slug, shortLabel, displayTitle
      },
      heroImage {
        asset,
        hotspot,
        crop,
        alt
      }
    }`,
  );

  return client.fetch(getUpcomingEventsQuery, { now, limit }, options);
}
```
